cmake_minimum_required(VERSION 3.16)
project(last_raven_launcher_preview C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_BUILD_TYPE Release)
get_filename_component(ROOT ../.. ABSOLUTE)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SDL REQUIRED IMPORTED_TARGET sdl2 SDL2_ttf)
file(GLOB RUNTIME_SOURCES "${ROOT}/tools/psprecomp/src/*.c" "${ROOT}/tools/psprecomp/src/hle/*.c")
add_library(runtime STATIC ${RUNTIME_SOURCES})
target_include_directories(runtime PUBLIC "${ROOT}/tools/psprecomp/include" "${DEPS}/include" "${FFMPEG}/include")
target_compile_definitions(runtime PRIVATE PSPRECOMP_HAVE_OPENH264=1 PSPRECOMP_HAVE_FFMPEG=1)
target_link_libraries(runtime PUBLIC "${DEPS}/lib/libopenh264.so" "${FFMPEG}/lib/libavcodec.so" "${FFMPEG}/lib/libavutil.so" m pthread)
add_executable(launcher "${ROOT}/host/launcher.c" "${ROOT}/host/settings.c" "${ROOT}/tools/psprecomp/src/host/settings.c")
target_link_libraries(launcher PRIVATE runtime PkgConfig::SDL)
add_executable(launcher-tests "${ROOT}/host/launcher_tests.c" "${ROOT}/host/settings.c" "${ROOT}/tools/psprecomp/src/host/settings.c")
target_link_libraries(launcher-tests PRIVATE runtime PkgConfig::SDL)
foreach(name settings-tool settings-tests)
    string(REPLACE "-" "_" source ${name})
    add_executable(${name} "${ROOT}/host/${source}.c" "${ROOT}/host/settings.c" "${ROOT}/tools/psprecomp/src/host/settings.c")
    target_include_directories(${name} PRIVATE "${ROOT}/tools/psprecomp/include")
    target_link_libraries(${name} PRIVATE m pthread)
endforeach()
# Assertions are the fixtures' checks; Release must not compile them out.
target_compile_options(launcher-tests PRIVATE -UNDEBUG)
target_compile_options(settings-tests PRIVATE -UNDEBUG)
foreach(name fps-tests fps-pose-tests)
    string(REPLACE "-" "_" source ${name})
    add_executable(${name} "${ROOT}/host/${source}.c")
    target_compile_options(${name} PRIVATE -UNDEBUG -Wall -Wextra -Werror)
    target_link_libraries(${name} PRIVATE m)
endforeach()
target_compile_options(launcher PRIVATE -Wall -Wextra -Werror)
add_executable(present-tests "${ROOT}/tools/psprecomp/tests/test_present.c" "${ROOT}/tools/psprecomp/src/host/save_dialog.c" "${ROOT}/tools/psprecomp/src/host/settings.c")
target_compile_definitions(present-tests PRIVATE HAVE_SDL2)
target_compile_options(present-tests PRIVATE -UNDEBUG -Wall -Wextra -Werror)
target_link_libraries(present-tests PRIVATE runtime PkgConfig::SDL)
add_executable(save-dialog-tests "${ROOT}/host/save_dialog_tests.c" "${ROOT}/tools/psprecomp/src/host/save_dialog.c"
    "${ROOT}/tools/psprecomp/src/host/present.c" "${ROOT}/host/render_gl.c" "${ROOT}/host/settings.c" "${ROOT}/tools/psprecomp/src/host/settings.c")
target_compile_definitions(save-dialog-tests PRIVATE HAVE_SDL2)
target_compile_options(save-dialog-tests PRIVATE -UNDEBUG -Wall -Wextra -Werror)
target_link_libraries(save-dialog-tests PRIVATE runtime PkgConfig::SDL)
add_executable(savedata-tests "${ROOT}/tools/psprecomp/tests/test_savedata.c")
target_compile_options(savedata-tests PRIVATE -UNDEBUG)
target_link_libraries(savedata-tests PRIVATE runtime)

set(RECOMP "${ROOT}/tools/psprecomp/tools/allegrexrecomp")
file(GLOB RECOMP_SOURCES "${RECOMP}/*.c" "${RECOMP}/crypto/*.c")
add_executable(allegrexrecomp ${RECOMP_SOURCES})
target_include_directories(allegrexrecomp PRIVATE "${RECOMP}")
target_link_libraries(allegrexrecomp PRIVATE runtime)
foreach(slug ac3p acsl aclr)
    # The window title now comes from the title's psp_title_info, compiled on
    # the device with its replacements, so these three archives are the same
    # objects; the player (psprecomp docs/PLAYER-LAYER.md stage 2) keeps one.
    # These are original host objects only. The user's generated game objects
    # and replacements requiring generated headers are linked on their device.
    add_library(host-${slug} STATIC "${ROOT}/host/boot.c" "${ROOT}/host/settings.c" "${ROOT}/tools/psprecomp/src/host/settings.c"
        "${ROOT}/tools/psprecomp/src/host/present.c" "${ROOT}/tools/psprecomp/src/host/save_dialog.c" "${ROOT}/host/render_gl.c"
        "${RECOMP}/loader.c" "${RECOMP}/container.c" "${RECOMP}/decode.c")
    target_include_directories(host-${slug} PRIVATE "${RECOMP}")
    target_compile_definitions(host-${slug} PRIVATE HAVE_SDL2)
    target_link_libraries(host-${slug} PRIVATE runtime PkgConfig::SDL)
endforeach()
